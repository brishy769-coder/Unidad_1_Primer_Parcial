Algoritmo TRESCALIFICACIONES
	Definir cal1, cal2, cal3, promedio Como Real
	Definir estado Como Cadena
	Escribir 'Ingrese la primera calificación: '
	Leer cal1
	Escribir 'Ingrese la segunda calificación: '
	Leer cal2
	Escribir 'Ingrese la tercera calificación: '
	Leer cal3
	promedio <- (cal1+cal2+cal3)/3
	Si promedio>=7 Entonces
		estado <- 'APROBADO'
	SiNo
		estado <- 'REPROBADO'
	FinSi
	Escribir 'Calificación 1: ', cal1
	Escribir 'Calificación 2: ', cal2
	Escribir 'Calificación 3: ', cal3
	Escribir 'Promedio: ', promedio
	Escribir 'Estado final: ', estado
FinAlgoritmo
